-- Prevent cycles in scope tree
CREATE OR REPLACE FUNCTION prevent_scope_cycle()
RETURNS TRIGGER AS $$
DECLARE
    current_id INT;
BEGIN
    current_id := NEW.parent_scope_id;

    WHILE current_id IS NOT NULL LOOP
        IF current_id = NEW.scope_id THEN
            RAISE EXCEPTION 'Cycle detected in scope tree';
        END IF;

        SELECT parent_scope_id INTO current_id
        FROM org_scope
        WHERE scope_id = current_id;
    END LOOP;

    RETURN NEW;
END;
$$ LANGUAGE plpgsql;

CREATE TRIGGER trg_scope_cycle
BEFORE INSERT OR UPDATE ON org_scope
FOR EACH ROW
EXECUTE FUNCTION prevent_scope_cycle();

-- FUNCTION: ADD NEW MEMBER TO ONGOING EVENTS

CREATE OR REPLACE FUNCTION add_member_to_ongoing_events()
RETURNS TRIGGER AS $$
BEGIN
    -- Only proceed if the new member has voter role
    IF NEW.is_voter = FALSE THEN
        RETURN NEW;
    END IF;

    INSERT INTO event_participants (event_id, orgid, uid)
    SELECT e.event_id, NEW.orgid, NEW.uid
    FROM events e
    WHERE e.orgid = NEW.orgid
      AND e.status = 'ACTIVE'
      AND e.start_time <= CURRENT_TIMESTAMP
      AND e.end_time   >= CURRENT_TIMESTAMP
      AND e.is_deleted = FALSE
      AND (
          -- Downward: event is at this scope or an ancestor scope
          (
              e.scope_only = FALSE
              AND e.scope_id IN (
                  SELECT scope_id FROM get_scope_ancestors(NEW.scope_id)
              )
          )
          OR
          -- Scope only: event scope must exactly match member scope
          (
              e.scope_only = TRUE
              AND e.scope_id = NEW.scope_id
          )
          OR
          -- Upward: event is in a descendant scope with upward flag
          (
              e.visibility_upward = TRUE
              AND e.scope_only = FALSE
              AND e.scope_id IN (
                  SELECT scope_id FROM get_scope_descendants(NEW.scope_id)
              )
          )
      )
    ON CONFLICT (event_id, orgid, uid) DO NOTHING;

    RETURN NEW;
END;
$$ LANGUAGE plpgsql;

CREATE TRIGGER trg_add_member_to_ongoing_events
AFTER INSERT ON member_roles
FOR EACH ROW
EXECUTE FUNCTION add_member_to_ongoing_events();

CREATE TRIGGER trg_add_member_to_ongoing_events_update
AFTER UPDATE OF is_voter ON member_roles
FOR EACH ROW
WHEN (OLD.is_voter = FALSE AND NEW.is_voter = TRUE)
EXECUTE FUNCTION add_member_to_ongoing_events();

-- Enforce identity validity
CREATE OR REPLACE FUNCTION validate_identity_wallet()
RETURNS TRIGGER AS $$
BEGIN
    IF NEW.identity_type = 'ORG' THEN
        -- org must exist
        IF NOT EXISTS (
            SELECT 1 FROM organization
            WHERE orgid = NEW.identity_id
        ) THEN
            RAISE EXCEPTION 'Invalid organization identity';
        END IF;

        -- uid must belong to org
        IF NOT EXISTS (
            SELECT 1 FROM org_members
            WHERE orgid = NEW.identity_id
              AND uid = NEW.uid
        ) THEN
            RAISE EXCEPTION 'UID does not belong to organization';
        END IF;

    ELSIF NEW.identity_type = 'GOV' THEN
        IF NOT EXISTS (
            SELECT 1 FROM gov_identity
            WHERE epic_id = NEW.identity_id
        ) THEN
            RAISE EXCEPTION 'Invalid government identity';
        END IF;
    END IF;

    RETURN NEW;
END;
$$ LANGUAGE plpgsql;

CREATE TRIGGER trg_validate_identity_wallet
BEFORE INSERT OR UPDATE ON identity_wallet
FOR EACH ROW
EXECUTE FUNCTION validate_identity_wallet();

    AND identity_type = 'ORG'

-- Only organizer can create event
CREATE OR REPLACE FUNCTION check_event_creator()
RETURNS TRIGGER AS $$
DECLARE
    user_scope INT;
BEGIN
    -- must be organizer
    IF NOT EXISTS (
        SELECT 1 FROM member_roles
        WHERE orgid = NEW.orgid
          AND uid = NEW.created_by_uid
          AND is_organizer = TRUE
    ) THEN
        RAISE EXCEPTION 'Only organizers can create events';
    END IF;

    -- get creator scope
    SELECT scope_id INTO user_scope
    FROM member_roles
    WHERE orgid = NEW.orgid
      AND uid = NEW.created_by_uid;

    -- enforce scope hierarchy
    IF NEW.scope_id NOT IN (
        SELECT scope_id
        FROM get_scope_descendants(user_scope)
    ) THEN
        RAISE EXCEPTION 'Cannot create event outside your scope';
    END IF;

    RETURN NEW;
END;
$$ LANGUAGE plpgsql;

CREATE TRIGGER trg_check_event_creator
BEFORE INSERT ON events
FOR EACH ROW
EXECUTE FUNCTION check_event_creator();

CREATE OR REPLACE FUNCTION maintain_closure_insert()
RETURNS TRIGGER AS $$
BEGIN
    -- Self-reference row
    INSERT INTO scope_closure (ancestor_id, descendant_id, depth)
    VALUES (NEW.scope_id, NEW.scope_id, 0);

    -- Connect all ancestors of parent to this new node
    IF NEW.parent_scope_id IS NOT NULL THEN
        INSERT INTO scope_closure (ancestor_id, descendant_id, depth)
        SELECT c.ancestor_id, NEW.scope_id, c.depth + 1
        FROM scope_closure c
        WHERE c.descendant_id = NEW.parent_scope_id;
    END IF;

    RETURN NEW;
END;
$$ LANGUAGE plpgsql;

CREATE TRIGGER trg_closure_insert
AFTER INSERT ON org_scope
FOR EACH ROW
EXECUTE FUNCTION maintain_closure_insert();


CREATE OR REPLACE FUNCTION get_scope_descendants(p_scope INT)
RETURNS TABLE(scope_id INT) AS $$
    SELECT descendant_id FROM scope_closure WHERE ancestor_id = p_scope;
$$ LANGUAGE sql;

CREATE OR REPLACE FUNCTION get_scope_ancestors(p_scope INT)
RETURNS TABLE(scope_id INT) AS $$
    SELECT ancestor_id FROM scope_closure WHERE descendant_id = p_scope;
$$ LANGUAGE sql;

CREATE OR REPLACE FUNCTION prevent_nonempty_scope_delete()
RETURNS TRIGGER AS $$
BEGIN
    IF EXISTS (
        SELECT 1 FROM org_scope
        WHERE parent_scope_id = OLD.scope_id
    ) THEN
        RAISE EXCEPTION
            'Cannot delete scope % — it has child scopes. Delete children first.',
            OLD.scope_name;
    END IF;

    IF EXISTS (
        SELECT 1 FROM member_roles
        WHERE scope_id = OLD.scope_id
    ) THEN
        RAISE EXCEPTION
            'Cannot delete scope % — members are assigned to it. Transfer members first.',
            OLD.scope_name;
    END IF;

    RETURN OLD;
END;
$$ LANGUAGE plpgsql;

CREATE TRIGGER trg_prevent_nonempty_scope_delete
BEFORE DELETE ON org_scope
FOR EACH ROW
EXECUTE FUNCTION prevent_nonempty_scope_delete();
-- =========================
-- FUNCTION: CHECK VOTING ELIGIBILITY
-- =========================
CREATE OR REPLACE FUNCTION get_visible_events(p_orgid VARCHAR, p_scope INT)
RETURNS TABLE(event_id INT) AS $$
SELECT e.event_id
FROM events e
WHERE e.orgid = p_orgid
AND e.is_deleted = FALSE
AND (
    -- Scope only: exact match only
    (
        e.scope_only = TRUE
        AND e.scope_id = p_scope
    )
    OR
    -- Downward visibility (default)
    (
        e.scope_only = FALSE
        AND e.scope_id IN (SELECT scope_id FROM get_scope_descendants(p_scope))
    )
    OR
    -- Upward visibility if allowed
    (
        e.scope_only = FALSE
        AND e.visibility_upward = TRUE
        AND e.scope_id IN (SELECT scope_id FROM get_scope_ancestors(p_scope))
    )
);
$$ LANGUAGE sql;

CREATE OR REPLACE FUNCTION check_vote_validity()
RETURNS TRIGGER AS $$
DECLARE ev RECORD;
DECLARE user_scope INT;
BEGIN
    SELECT * INTO ev FROM events WHERE event_id = NEW.event_id;

    IF CURRENT_TIMESTAMP < ev.start_time OR CURRENT_TIMESTAMP > ev.end_time THEN
        RAISE EXCEPTION 'Voting not active';
    END IF;

    SELECT scope_id INTO user_scope
    FROM member_roles
    WHERE orgid = NEW.orgid AND uid = NEW.uid;

    -- scope_only: must be exact match
    IF ev.scope_only = TRUE THEN
        IF user_scope <> ev.scope_id THEN
            RAISE EXCEPTION 'Not eligible';
        END IF;
        RETURN NEW;
    END IF;

    -- Default: check descendants, and optionally ancestors
    IF user_scope NOT IN (
        SELECT scope_id FROM get_scope_descendants(ev.scope_id)
    )
    AND NOT (
        ev.visibility_upward = TRUE
        AND user_scope IN (
            SELECT scope_id FROM get_scope_ancestors(ev.scope_id)
        )
    ) THEN
        RAISE EXCEPTION 'Not eligible';
    END IF;

    RETURN NEW;
END;
$$ LANGUAGE plpgsql;

CREATE TRIGGER trg_vote_validity
BEFORE INSERT ON votes
FOR EACH ROW
EXECUTE FUNCTION check_vote_validity();

-- =========================
-- FUNCTION: AUTO ADD PARTICIPANTS ON EVENT CREATION
-- =========================
CREATE OR REPLACE FUNCTION populate_event_participants()
RETURNS TRIGGER AS $$
BEGIN
    INSERT INTO event_participants (event_id, orgid, uid)
    SELECT NEW.event_id, om.orgid, om.uid
    FROM org_members om
    JOIN member_roles mr ON mr.orgid = om.orgid AND mr.uid = om.uid
    WHERE om.orgid = NEW.orgid AND mr.is_voter = TRUE
      AND (
          -- Scope only: exact match, no descendants
          (
              NEW.scope_only = TRUE
              AND mr.scope_id = NEW.scope_id
          )
          OR
          -- Downward: voters in descendant scopes (default)
          (
              NEW.scope_only = FALSE
              AND mr.scope_id IN (
                  SELECT scope_id FROM get_scope_descendants(NEW.scope_id)
              )
          )
          OR
          -- Upward: ancestor scopes, only if flag is set
          (
              NEW.visibility_upward = TRUE
              AND NEW.scope_only = FALSE
              AND mr.scope_id IN (
                  SELECT scope_id FROM get_scope_ancestors(NEW.scope_id)
              )
          )
      )
    ON CONFLICT (event_id, orgid, uid) DO NOTHING;

    RETURN NEW;
END;
$$ LANGUAGE plpgsql;

CREATE TRIGGER trg_populate_event_participants
AFTER INSERT ON events
FOR EACH ROW
EXECUTE FUNCTION populate_event_participants();

-- 1. ANONYMIZED VOTING
-- =========================

-- Added hashed voter identity (instead of exposing uid directly)

-- Function to generate hash
CREATE OR REPLACE FUNCTION generate_voter_hash(p_event INT, p_org VARCHAR, p_uid VARCHAR)
RETURNS TEXT AS $$
BEGIN
    RETURN encode(
        digest(
            p_event::TEXT || '-' || p_org || '-' || p_uid
            || '-' || current_setting('app.voter_hash_salt'),
            'sha256'
        ),
        'hex'
    );
END;
$$ LANGUAGE plpgsql;

-- Trigger: replace uid visibility with hash
CREATE OR REPLACE FUNCTION anonymize_vote()
RETURNS TRIGGER AS $$
BEGIN
    NEW.voter_hash := generate_voter_hash(NEW.event_id, NEW.orgid, NEW.uid);
    RETURN NEW;
END;
$$ LANGUAGE plpgsql;

CREATE TRIGGER trg_anonymize_vote
BEFORE INSERT ON votes
FOR EACH ROW
EXECUTE FUNCTION anonymize_vote();


-- =========================
-- 2. FRAUD DETECTION (IP + DEVICE)
-- =========================

-- Extend votes table
ALTER TABLE votes
ADD COLUMN ip_address INET,
ADD COLUMN device_fingerprint TEXT;

-- Fraud log table
CREATE TABLE fraud_logs (
    fraud_id SERIAL PRIMARY KEY,
    event_id INT,
    orgid VARCHAR(20),
    uid VARCHAR(20),
    ip_address INET,
    device_fingerprint TEXT,
    reason TEXT,
    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP
);

-- Function: detect suspicious activity
CREATE OR REPLACE FUNCTION detect_vote_fraud()
RETURNS TRIGGER AS $$
DECLARE
    ip_count INT;
    device_count INT;
BEGIN
    -- Count votes from same IP in same event
    SELECT COUNT(*) INTO ip_count
    FROM votes
    WHERE event_id = NEW.event_id
      AND ip_address = NEW.ip_address;

    -- Count votes from same device
    SELECT COUNT(*) INTO device_count
    FROM votes
    WHERE event_id = NEW.event_id
      AND device_fingerprint = NEW.device_fingerprint;

    IF ip_count > 4 THEN
        INSERT INTO fraud_logs(event_id, orgid, uid, ip_address, device_fingerprint, reason)
        VALUES (NEW.event_id, NEW.orgid, NEW.uid, NEW.ip_address, NEW.device_fingerprint,
                'Too many votes from same IP');
    END IF;

    IF device_count > 3 THEN
        INSERT INTO fraud_logs(event_id, orgid, uid, ip_address, device_fingerprint, reason)
        VALUES (NEW.event_id, NEW.orgid, NEW.uid, NEW.ip_address, NEW.device_fingerprint,
                'Multiple votes from same device');
    END IF;

    RETURN NEW;
END;
$$ LANGUAGE plpgsql;

CREATE TRIGGER trg_detect_vote_fraud
AFTER INSERT ON votes
FOR EACH ROW
EXECUTE FUNCTION detect_vote_fraud();


-- =========================
-- 3. AUDIT LOGGING SYSTEM
-- =========================

CREATE TABLE audit_logs (
    log_id SERIAL PRIMARY KEY,
    table_name VARCHAR(50),
    operation VARCHAR(10), -- INSERT, UPDATE, DELETE
    record_id TEXT,
    changed_data JSONB,
    changed_by VARCHAR(50),
    changed_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP
);

-- Generic audit function
CREATE OR REPLACE FUNCTION audit_trigger_function()
RETURNS TRIGGER AS $$
DECLARE
    data JSONB;
    rec_id TEXT;
BEGIN
    IF TG_OP = 'DELETE' THEN
        data := to_jsonb(OLD);
        rec_id := OLD::TEXT;
    ELSE
        data := to_jsonb(NEW);
        rec_id := NEW::TEXT;
    END IF;

    INSERT INTO audit_logs(table_name, operation, record_id, changed_data, changed_by)
    VALUES (TG_TABLE_NAME, TG_OP, rec_id, data, current_user);

    IF TG_OP = 'DELETE' THEN
        RETURN OLD;
    END IF;
    RETURN NEW;
END;
$$ LANGUAGE plpgsql;

-- Attach audit triggers to critical tables

CREATE TRIGGER audit_events
AFTER INSERT OR UPDATE OR DELETE ON events
FOR EACH ROW EXECUTE FUNCTION audit_trigger_function();

CREATE TRIGGER audit_votes
AFTER INSERT OR UPDATE OR DELETE ON votes
FOR EACH ROW EXECUTE FUNCTION audit_trigger_function();

CREATE TRIGGER audit_org_members
AFTER INSERT OR UPDATE OR DELETE ON org_members
FOR EACH ROW EXECUTE FUNCTION audit_trigger_function();

CREATE TRIGGER audit_member_roles
AFTER INSERT OR UPDATE OR DELETE ON member_roles
FOR EACH ROW EXECUTE FUNCTION audit_trigger_function();

-- =========================
-- FUNCTION: REFRESH PARTICIPANTS ON VISIBILITY CHANGE
-- =========================
CREATE OR REPLACE FUNCTION refresh_event_participants_on_visibility_change()
RETURNS TRIGGER AS $$
BEGIN
    -- -------------------------------------------------------
    -- STEP 1: Remove participants who no longer qualify
    --         SAFETY: never touch rows where has_voted = TRUE
    -- -------------------------------------------------------
    DELETE FROM event_participants ep
    WHERE ep.event_id = NEW.event_id
      AND ep.has_voted = FALSE
      AND NOT EXISTS (
          SELECT 1
          FROM member_roles mr
          WHERE mr.orgid = ep.orgid
            AND mr.uid   = ep.uid
            AND mr.is_voter = TRUE
            AND (
                -- scope_only ON: exact scope match only
                (
                    NEW.scope_only = TRUE
                    AND mr.scope_id = NEW.scope_id
                )
                OR
                -- Default downward: voter is in a descendant scope
                (
                    NEW.scope_only = FALSE
                    AND mr.scope_id IN (
                        SELECT scope_id FROM get_scope_descendants(NEW.scope_id)
                    )
                )
                OR
                -- Upward: voter is in an ancestor scope, flag must be on
                (
                    NEW.scope_only = FALSE
                    AND NEW.visibility_upward = TRUE
                    AND mr.scope_id IN (
                        SELECT scope_id FROM get_scope_ancestors(NEW.scope_id)
                    )
                )
            )
      );

    -- -------------------------------------------------------
    -- STEP 2: Add participants who newly qualify
    -- -------------------------------------------------------
    INSERT INTO event_participants (event_id, orgid, uid)
    SELECT NEW.event_id, om.orgid, om.uid
    FROM org_members om
    JOIN member_roles mr
      ON mr.orgid = om.orgid
     AND mr.uid   = om.uid
    WHERE om.orgid      = NEW.orgid
      AND om.is_deleted = FALSE
      AND mr.is_voter   = TRUE
      AND (
          -- scope_only ON: exact scope match only
          (
              NEW.scope_only = TRUE
              AND mr.scope_id = NEW.scope_id
          )
          OR
          -- Default downward: voter is in a descendant scope
          (
              NEW.scope_only = FALSE
              AND mr.scope_id IN (
                  SELECT scope_id FROM get_scope_descendants(NEW.scope_id)
              )
          )
          OR
          -- Upward: voter is in an ancestor scope, flag must be on
          (
              NEW.scope_only = FALSE
              AND NEW.visibility_upward = TRUE
              AND mr.scope_id IN (
                  SELECT scope_id FROM get_scope_ancestors(NEW.scope_id)
              )
          )
      )
    ON CONFLICT (event_id, orgid, uid) DO NOTHING;

    RETURN NEW;
END;
$$ LANGUAGE plpgsql;

-- Fires only when the three relevant columns are touched
CREATE TRIGGER trg_refresh_event_participants_on_visibility
AFTER UPDATE OF visibility_upward, scope_only, scope_id ON events
FOR EACH ROW
WHEN (
    OLD.visibility_upward IS DISTINCT FROM NEW.visibility_upward
    OR OLD.scope_only     IS DISTINCT FROM NEW.scope_only
    OR OLD.scope_id       IS DISTINCT FROM NEW.scope_id
)
EXECUTE FUNCTION refresh_event_participants_on_visibility_change();

CREATE OR REPLACE FUNCTION block_vote_on_deleted_event()
RETURNS TRIGGER AS $$
BEGIN
    IF EXISTS (
        SELECT 1 FROM events
        WHERE event_id = NEW.event_id
          AND is_deleted = TRUE
    ) THEN
        RAISE EXCEPTION 'Cannot vote on a deleted event';
    END IF;
    RETURN NEW;
END;
$$ LANGUAGE plpgsql;

CREATE TRIGGER trg_block_vote_deleted_event
BEFORE INSERT ON votes
FOR EACH ROW
EXECUTE FUNCTION block_vote_on_deleted_event();
